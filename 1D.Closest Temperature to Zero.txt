#include <stdio.h>
#include <stdlib.h>
int main() {
    int N,i,closest;
    scanf("%d",&N);
    int arr[N];
    for(i=0;i<N;i++) scanf("%d",&arr[i]);
    closest=arr[0];
    for(i=1;i<N;i++) {
        if(abs(arr[i])<abs(closest) || (abs(arr[i])==abs(closest) && arr[i]>closest))
            closest=arr[i];
    }
    printf("%d",closest);
    return 0;
}
